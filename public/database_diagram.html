    users {
        bigint id PK
        string name
        string login UK
        string email UK
        string role
        string password
        timestamp email_verified_at
    }

    members {
        bigint id PK
        bigint user_id FK
        string full_name
        string email UK
        string phone
        string checkin_code UK
        string profile_photo_path
        tinyint photo_change_count
        date joined_at
        date expires_at
    }

    membership_plans {
        bigint id PK
        string name UK
        tinyint duration_months
        integer price
        boolean is_active
    }

    membership_subscriptions {
        bigint id PK
        bigint member_id FK
        bigint membership_plan_id FK
        date start_date
        date end_date
        integer amount_paid
        string payment_method
        string status
    }

    daily_guests {
        bigint id PK
        string full_name
        string phone
        string visit_type
        timestamp visit_at
    }

    checkins {
        bigint id PK
        bigint member_id FK
        bigint daily_guest_id FK
        timestamp checked_in_at
        string checkin_method
        string verification_status
        bigint verified_by FK
    }

    categories {
        bigint id PK
        string name UK
        text description
    }

    products {
        bigint id PK
        bigint category_id FK
        string name
        string sku UK
        integer price
        integer stock
        string unit
        boolean is_active
    }

    product_stock_logs {
        bigint id PK
        bigint product_id FK
        enum type
        integer quantity
        string description
        string reference_id
        bigint user_id FK
    }

    transactions {
        bigint id PK
        string invoice UK
        string snap_token
        bigint member_id FK
        bigint daily_guest_id FK
        bigint cashier_user_id FK
        string type
        integer amount
        string payment_method
        string payment_status
        timestamp transaction_at
    }

    transaction_items {
        bigint id PK
        bigint transaction_id FK
        bigint product_id FK
        string item_name
        integer quantity
        integer unit_price
        integer subtotal
    }

    expense_records {
        bigint id PK
        string title
        string category
        integer amount
        string payment_method
        date expense_date
    }

    announcements {
        bigint id PK
        string title
        text body
        string status
        timestamp publish_at
    }

    member_feedbacks {
        bigint id PK
        bigint user_id FK
        bigint member_id FK
        string name
        string subject
        text message
    }

    membership_renewal_requests {
        bigint id PK
        bigint member_id FK
        bigint user_id FK
        bigint membership_plan_id FK
        integer amount
        string payment_method
        string status
        bigint reviewed_by FK
    }

    profile_photo_change_requests {
        bigint id PK
        bigint user_id FK
        bigint member_id FK
        string requested_photo_path
        string status
        bigint reviewed_by FK
    }
